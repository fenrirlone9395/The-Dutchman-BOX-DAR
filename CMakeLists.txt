cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
set(CMAKE_OSX_ARCHITECTURES "arm64")
# Explicit SDK headers also support Command Line Tools installations without toolchain libc++ headers.
execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE RACK_SDK OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(CMAKE_OSX_SYSROOT "${RACK_SDK}")
set(CMAKE_CXX_FLAGS_INIT "-isystem ${RACK_SDK}/usr/include/c++/v1")
set(CMAKE_OBJCXX_FLAGS_INIT "-isystem ${RACK_SDK}/usr/include/c++/v1")
project(DesktopAudioRack LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
add_executable(DesktopAudioRack MACOSX_BUNDLE src/Main.mm src/Output.mm src/ProcessTap.mm src/MeterView.mm src/RackControls.mm src/SpatialView.mm src/Theme.mm src/plugins/AURack.mm src/plugins/AURackUI.mm)
target_compile_options(DesktopAudioRack PRIVATE -fobjc-arc -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(DesktopAudioRack PRIVATE "-framework CoreAudioKit" "-framework Cocoa" "-framework ScreenCaptureKit" "-framework CoreMedia" "-framework AudioToolbox" "-framework CoreAudio")
set_target_properties(DesktopAudioRack PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
add_custom_command(TARGET DesktopAudioRack POST_BUILD COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:DesktopAudioRack>")
enable_testing()
add_executable(bridge_tests tests/bridge_tests.cpp)
add_test(NAME bridge_tests COMMAND bridge_tests)
add_executable(dsp_tests tests/dsp_tests.cpp)
add_test(NAME dsp_tests COMMAND dsp_tests)
# Keep assertions active in Release test targets.
target_compile_options(bridge_tests PRIVATE -UNDEBUG)
target_compile_options(dsp_tests PRIVATE -UNDEBUG)
add_executable(au_tests tests/au_tests.mm src/plugins/AURack.mm)
target_compile_options(au_tests PRIVATE -fobjc-arc -UNDEBUG)
target_link_libraries(au_tests PRIVATE "-framework Foundation" "-framework AudioToolbox")
add_test(NAME au_tests COMMAND au_tests)

add_executable(spatial_tests tests/spatial_tests.cpp)
target_compile_options(spatial_tests PRIVATE -UNDEBUG)
add_test(NAME spatial_tests COMMAND spatial_tests)

add_executable(tap_input_tests tests/tap_input_tests.mm)
target_compile_options(tap_input_tests PRIVATE -fobjc-arc -UNDEBUG)
target_link_libraries(tap_input_tests PRIVATE "-framework Foundation" "-framework CoreAudio")
add_test(NAME tap_input_tests COMMAND tap_input_tests)
